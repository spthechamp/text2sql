CREATE DATABASE employees;

CREATE TABLE departments (
	dept_no CHAR(4) NOT NULL, 
	dept_name VARCHAR(40) NOT NULL, 
	PRIMARY KEY (dept_no)
)COLLATE utf8mb4_0900_ai_ci DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB

/*
3 rows from departments table:
dept_no	dept_name
A	Technology
B	Technology
C	HR
*/


CREATE TABLE dept_emp (
	emp_no INTEGER NOT NULL, 
	dept_no CHAR(4) NOT NULL, 
	from_date DATE, 
	to_date DATE, 
	PRIMARY KEY (emp_no, dept_no)
)COLLATE utf8mb4_0900_ai_ci DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB

/*
3 rows from dept_emp table:
emp_no	dept_no	from_date	to_date
1	A	1990-01-01	None
2	B	1991-02-01	2010-06-30
3	C	1992-03-01	None
*/


CREATE TABLE dept_manager (
	dept_no CHAR(4) NOT NULL, 
	emp_no INTEGER NOT NULL, 
	from_date DATE, 
	to_date DATE, 
	PRIMARY KEY (dept_no, emp_no)
)COLLATE utf8mb4_0900_ai_ci DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB

/*

CREATE TABLE emp (
	emp_no INTEGER NOT NULL, 
	birth_date DATE, 
	first_name VARCHAR(14), 
	last_name VARCHAR(16), 
	gender ENUM('M','F'), 
	hire_date DATE, 
	PRIMARY KEY (emp_no)
)COLLATE utf8mb4_0900_ai_ci DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB

/*
3 rows from emp table:

2	1967-01-01	Thor	Bjornson	M	1990-01-01
3	1968-01-01	Ronda	Rowsey	F	1990-01-01
*/


CREATE TABLE personal (
	emp_no INTEGER NOT NULL, 
	city VARCHAR(45) NOT NULL, 
	dependent INTEGER, 
	age INTEGER NOT NULL, 
	contact INTEGER NOT NULL, 
	PRIMARY KEY (emp_no)
)COLLATE utf8mb4_0900_ai_ci DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB

/*

emp_no	city	dependent	age	contact

CREATE TABLE salaries (
	emp_no INTEGER NOT NULL, 
	salary INTEGER NOT NULL, 
	from_date DATE, 
	to_date DATE, 
	PRIMARY KEY (emp_no)
)COLLATE utf8mb4_0900_ai_ci DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB

CREATE TABLE titles (
	emp_no INTEGER NOT NULL,
